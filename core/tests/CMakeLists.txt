# Foundation file services are tested without enabling the AI runtime.
add_executable(test_filesystem test_filesystem.cpp)
target_link_libraries(test_filesystem PRIVATE CVCoreLib)
if(NOT MSVC)
    target_compile_options(test_filesystem PRIVATE -fPIC)
    if(NOT APPLE)
        target_link_options(test_filesystem PRIVATE -no-pie)
    endif()
endif()
add_test(NAME test_filesystem COMMAND test_filesystem)
set_tests_properties(test_filesystem PROPERTIES LABELS "fast;common;filesystem")
cloudviewer_set_test_runtime_environment(test_filesystem)
# Include this dependency in the existing consumer gate as well.
if(TARGET aicore-fast-tests)
    add_dependencies(aicore-fast-tests test_filesystem)
endif()

add_executable(test_parallel test_parallel.cpp)
target_link_libraries(test_parallel PRIVATE CVCoreLib)
if(WITH_OPENMP)
    find_package(OpenMP QUIET COMPONENTS CXX)
    if(TARGET OpenMP::OpenMP_CXX)
        target_link_libraries(test_parallel PRIVATE OpenMP::OpenMP_CXX)
    endif()
endif()
if(NOT MSVC)
    target_compile_options(test_parallel PRIVATE -fPIC)
    if(NOT APPLE)
        target_link_options(test_parallel PRIVATE -no-pie)
    endif()
endif()
add_test(NAME test_parallel COMMAND test_parallel)
set_tests_properties(test_parallel PROPERTIES LABELS "fast;common;parallel")
cloudviewer_set_test_runtime_environment(test_parallel)
if(TARGET aicore-fast-tests)
    add_dependencies(aicore-fast-tests test_parallel)
endif()
