# Foundation file services are tested without enabling the AI runtime.
if(NOT TARGET gtest_main)
    include(${CloudViewer_3RDPARTY_DIR}/googletest/googletest.cmake)
endif()
add_executable(test_filesystem test_filesystem.cpp)
target_link_libraries(test_filesystem PRIVATE CVCoreLib gtest_main)
if(NOT MSVC)
    target_compile_options(test_filesystem PRIVATE -fPIC)
    if(NOT APPLE)
        target_link_options(test_filesystem PRIVATE -no-pie)
    endif()
endif()
cloudviewer_add_test(NAME test_filesystem COMMAND test_filesystem)
set_tests_properties(test_filesystem PROPERTIES LABELS "fast;common;filesystem")
cloudviewer_set_test_runtime_environment(test_filesystem)

add_executable(test_parallel test_parallel.cpp)
target_link_libraries(test_parallel PRIVATE CVCoreLib gtest_main)
if(WITH_OPENMP)
    find_package(OpenMP QUIET COMPONENTS CXX)
    if(TARGET OpenMP::OpenMP_CXX)
        target_link_libraries(test_parallel PRIVATE OpenMP::OpenMP_CXX)
    endif()
endif()
if(NOT MSVC)
    target_compile_options(test_parallel PRIVATE -fPIC)
    if(NOT APPLE)
        target_link_options(test_parallel PRIVATE -no-pie)
    endif()
endif()
cloudviewer_add_test(NAME test_parallel COMMAND test_parallel)
set_tests_properties(test_parallel PROPERTIES LABELS "fast;common;parallel")
cloudviewer_set_test_runtime_environment(test_parallel)

# Exercise QtCompat's parser boundaries without the application or AI runtime.

    target_include_directories(test_qt_compat PRIVATE "${CMAKE_SOURCE_DIR}/core/include")

        Qt::Widgets gtest_main ${QT_COMPAT_PLATFORM_LIBRARIES})
    if(NOT MSVC)
        target_compile_options(test_qt_compat PRIVATE -fPIC)
    endif()
    cloudviewer_add_test(NAME test_qt_compat COMMAND test_qt_compat)
    set_tests_properties(test_qt_compat PROPERTIES LABELS "fast;common;qt-compat")
    cloudviewer_set_test_runtime_environment(test_qt_compat)
endif()
