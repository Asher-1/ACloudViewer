cmake_minimum_required(VERSION 3.24)
project(CloudViewerTestContracts LANGUAGES CXX)
enable_testing()
option(BUILD_UNIT_TESTS "Exercise the unified runner" ON)
add_custom_target(check-test-system)
set(CLOUDVIEWER_TEST_CONTRACT_FIXTURE TRUE)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${ROOT}/tests/Testing.cmake")
if(NOT CASE)
    set(CASE valid)
endif()
if(CASE STREQUAL model-disabled)
    set(AICORE_TEST_ENABLE_MODEL_TESTS OFF)
else()
    set(AICORE_TEST_ENABLE_MODEL_TESTS ON)
endif()
set(AICORE_TIER_FAST_EXCLUDE "model|gpu|e2e|cvpluginapi|network|graphics|benchmark|slow" CACHE INTERNAL
    "Fixture fast-tier exclusions" FORCE)
set(AICORE_TIER_CONTRACT_LABEL "capi" CACHE INTERNAL
    "Fixture contract-tier label" FORCE)
set(AICORE_TIER_CONTRACT_MODEL_EXCLUDE "model|gpu|e2e" CACHE INTERNAL
    "Fixture contract-tier exclusions" FORCE)
set(AICORE_TIER_MODEL_LABEL "model" CACHE INTERNAL
    "Fixture model-tier label" FORCE)

add_custom_target(aicore-fast-tests)
add_custom_target(aicore-contract-tests)
add_custom_target(aicore-model-tests)
add_subdirectory(owner)
if(NOT CASE STREQUAL without-app-registration AND NOT CASE STREQUAL empty
        AND NOT CASE STREQUAL graphics-only AND NOT CASE STREQUAL script-only)
    add_subdirectory(app/tests)
endif()
# This target deliberately appears after owner/ registers its test.
if(NOT CASE STREQUAL missing AND NOT CASE STREQUAL empty)
    add_executable(forward_target pass.cpp)
endif()
# These executables only prove compiler/prerequisite wiring, not C++ assertions.
foreach(probe direct_target wrapper_target graphics_target network_target forward_target app_registration_probe)
    if(TARGET ${probe})
        set_property(TARGET ${probe} PROPERTY CLOUDVIEWER_TEST_KIND integration)
    endif()
endforeach()
if(CASE STREQUAL gtest)
    # Synthetic targets exercise transitive aliases/generator expressions;
    # assertion semantics belong to the real GoogleTest suites, not this fixture.
    add_library(gtest INTERFACE)
    add_library(test_framework INTERFACE)
    target_link_libraries(test_framework INTERFACE "$<LINK_ONLY:gtest>")
    add_library(Fixture::Framework ALIAS test_framework)
    add_executable(framed_unit pass.cpp)
    target_link_libraries(framed_unit PRIVATE Fixture::Framework)
    cloudviewer_add_test(NAME fixture-gtest COMMAND framed_unit LABELS unit)
endif()
if(CASE STREQUAL nongtest)
    add_executable(unframed_unit pass.cpp)
    cloudviewer_add_test(NAME fixture-nongtest COMMAND unframed_unit LABELS unit)
endif()
add_subdirectory("${ROOT}/tests" root-tests)

if(CASE STREQUAL valid OR CASE STREQUAL model-disabled)
    get_target_property(all_deps cpp-tests MANUALLY_ADDED_DEPENDENCIES)
    foreach(required direct_target wrapper_target forward_target graphics_target app_registration_probe)
        if(NOT required IN_LIST all_deps)
            message(FATAL_ERROR "cpp-tests closure omitted target '${required}'")
        endif()
    endforeach()
    get_target_property(ci_deps cpp-tests-ci MANUALLY_ADDED_DEPENDENCIES)
    if(NOT wrapper_target IN_LIST ci_deps)
        message(FATAL_ERROR "wrapper target was not captured in cpp-tests-ci closure")
    endif()
    foreach(forbidden direct_target forward_target graphics_target)
        if(forbidden IN_LIST ci_deps)
            message(FATAL_ERROR "non-CI prerequisite '${forbidden}' leaked into cpp-tests-ci")
        endif()
    endforeach()
    if(NOT app_registration_probe IN_LIST ci_deps)
        message(FATAL_ERROR "app registration prerequisite omitted from automatic CI closure")
    endif()
    get_target_property(graphics_deps cpp-tests-graphics MANUALLY_ADDED_DEPENDENCIES)
    foreach(required graphics_target wrapper_target)
        if(NOT required IN_LIST graphics_deps)
            message(FATAL_ERROR "graphics closure omitted ${required}")
        endif()
    endforeach()
    get_target_property(fast_deps cpp-tests-aicore-fast MANUALLY_ADDED_DEPENDENCIES)
    if(NOT wrapper_target IN_LIST fast_deps)
        message(FATAL_ERROR "AICore fast closure omitted wrapper target")
    endif()
endif()
if((CASE STREQUAL empty OR CASE STREQUAL graphics-only) AND TARGET check-cpp)
    message(FATAL_ERROR "empty cpp-ci selection created check-cpp")
endif()
